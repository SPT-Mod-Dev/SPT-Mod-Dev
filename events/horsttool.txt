country_event = {
    id = hosttool.1000
    title = "Move Host Tool Confirmation"
    # each popup names its own request; the global fallback covers a popup without a copy
    desc = { text = hosttool.1000.desc_req trigger = { has_event_target = ht_req_token } }
    desc = { text = hosttool.1000.desc trigger = { NOT = { has_event_target = ht_req_token } } }
    picture = GFX_report_event_001
    is_triggered_only = yes

    option = {
        name = "Yes, move the host tool"
        ai_chance = { factor = 0 } # an AI host (dropped player) must never hand the tool away
        # ht_new_host was captured as an ENGINE-MANAGED event target at click time
        # (change_host_nation_click). The engine invalidates it if the country died
        # while this popup sat open, so a stale confirm is a safe NO-OP - the raw
        # temp_new_host object var is never dereferenced here, and host flags are
        # only cleared once the target is proven alive.
        # Only the CURRENT host's live request may act (ht_host_move_request_is_live): a popup
        # orphaned by Clear All or a handoff can never move the tool for a newer request.
        if = {
            limit = { ht_host_move_request_is_live = yes }
            if = {
                limit = { event_target:ht_new_host = { exists = yes } }
                # Full privilege handoff: host flags AND all host-only admin state,
                # so the former host keeps no working picker/decision categories.
                every_country = {
                    clr_country_flag = host
                    clr_country_flag = pause_blocked
                    clr_country_flag = transfer_enabled
                    clr_country_flag = ht_focus_unc
                    clr_country_flag = ht_focus_com
                    clr_country_flag = ht_law_mode
                    clr_country_flag = ht_spirit_add
                    clr_country_flag = ht_spirit_remove
                    clr_country_flag = ht_annex_pending
                    clr_country_flag = ht_host_move_pending
                }
                clear_variable = global.ht_focus_state
                clear_global_event_target = ht_focus_state
                # the old host's open annex request dies with its authority
                clear_global_event_target = ht_annexed_country
                clear_global_event_target = ht_annexer_country
                every_state = {
                    limit = { OR = { has_state_flag = ht_live_annex_req has_state_flag = ht_live_move_req } }
                    clr_state_flag = ht_live_annex_req
                    clr_state_flag = ht_live_move_req
                }
                # Old host's marked states must not leak into the new host's session
                for_each_scope_loop = {
                    array = marked_states
                    clear_variable = mark_status
                }
                clear_array = marked_states
                event_target:ht_new_host = {
                    set_country_flag = host
                    set_country_flag = transfer_enabled
                    # notification copy: the broadcast renders after ht_new_host is cleared
                    save_global_event_target_as = ht_note_new_host
                }
                # Send ONE global notification
                news_event = { id = hosttool.1001 }
            }
            # else: target died before the confirm - keep the current host untouched
            clr_country_flag = ht_host_move_pending
            every_state = { limit = { has_state_flag = ht_live_move_req } clr_state_flag = ht_live_move_req }
            clear_global_event_target = ht_new_host
        }
        clear_variable = temp_new_host
        # this popup is answered: free its token, live or orphaned
        ht_release_request_token = yes
    }

    option = {
        name = "No, cancel"
        ai_chance = { factor = 1 }
        # cancels only its own live request, never a newer one
        if = {
            limit = { ht_host_move_request_is_live = yes }
            clr_country_flag = ht_host_move_pending
            every_state = { limit = { has_state_flag = ht_live_move_req } clr_state_flag = ht_live_move_req }
            clear_global_event_target = ht_new_host
        }
        clear_variable = temp_new_host
        # this popup is answered: free its token, live or orphaned
        ht_release_request_token = yes
    }
}

# Fix AI control: hidden return event, fired 2h after the player was parked on SOL. It is
# scheduled on EVERY existing country (fix_ai_control_click) so the cleanup below runs even if
# the host, MTF or the glitched country disappears meanwhile; whichever copy runs first does
# the work and releases the guard, the others find it released and do nothing.
country_event = {
    id = hosttool.60
    title = "Fix AI Control"
    desc = "Returning player control"
    picture = GFX_report_event_001
    hidden = yes
    is_triggered_only = yes

    option = {
        name = "Return the player"
        if = {
            limit = { has_variable = global.ai_fix_return_country }
            if = {
                limit = { event_target:ht_aifix_country = { exists = yes } }
                event_target:ht_aifix_country = { change_tag_from = SOL }

            # which is also the observer tag, and the guards below are released.
            clear_variable = global.ai_fix_return_country
            clear_global_event_target = ht_aifix_country
        }

# Annex confirmation (fired to the host; vars target/taker were set by annex_country_click)
country_event = {
    id = hosttool.63
    title = "Annex Confirmation"
    # each popup names its own request; the global fallback covers a popup without a copy
    desc = { text = hosttool.63.desc_req trigger = { has_event_target = ht_req_token } }
    desc = { text = hosttool.63.desc trigger = { NOT = { has_event_target = ht_req_token } } }
    picture = GFX_report_event_001
    is_triggered_only = yes

    option = {
        name = "Yes, annex them"
        ai_chance = { factor = 0 } # an AI host (dropped player) must never confirm an annex
        # Only the CURRENT host's live request may act (ht_annex_request_is_live): a popup
        # orphaned by Clear All or a host handoff can never annex or clear a newer request.
        if = {
            limit = { ht_annex_request_is_live = yes }
            clr_country_flag = ht_annex_pending
            every_state = { limit = { has_state_flag = ht_live_annex_req } clr_state_flag = ht_live_annex_req }
            # AMSB: both sides were captured as ENGINE-MANAGED global event targets at click time
            # (annex_country_click). The engine invalidates them if either country dies before this
            # option runs, so the liveness guard turns a stale confirm into a safe no-op instead of
            # a use-after-free. Raw object vars (ROOT.target/taker) are never dereferenced here.
            if = {
                limit = {
                    event_target:ht_annexed_country = { exists = yes NOT = { tag = ROOT } }
                    event_target:ht_annexer_country = { exists = yes }
                }
                # crash-correlation breadcrumb: the annex EXECUTES here, not at click time
                # AMSB 2026-08-12: annex also CORES the annexed land for the annexer (user request).
                # Runs before annex_country while the target still owns its states; event targets
                # only, same liveness guard as the annex itself (add_core_of = event_target: has
                # in-tree precedent at common/decisions/RAJ_GOE.txt:726).
                event_target:ht_annexed_country = {
                    every_owned_state = {
                        add_core_of = event_target:ht_annexer_country
                    }
                    # notification copy: the broadcast renders after the operation targets
                    # below are cleared
                    save_global_event_target_as = ht_note_annexed
                }
                event_target:ht_annexer_country = { save_global_event_target_as = ht_note_annexer }
                log = "HOSTTOOL: annex_executed [ht_annexed_country.GetName] -> [ht_annexer_country.GetName]"
                event_target:ht_annexer_country = {
                    annex_country = { target = event_target:ht_annexed_country transfer_troops = yes }
                }
                news_event = { id = hosttool.64 }
            }
            # else: stale/dead confirm, annex skipped
            clear_global_event_target = ht_annexed_country
            clear_global_event_target = ht_annexer_country
            for_each_scope_loop = {
                array = marked_states
                clear_variable = mark_status
            }
            clear_array = marked_states
            clear_variable = target
            clear_variable = taker
            clear_variable = first_marked_state
        }
        # this popup is answered: free its token, live or orphaned
        ht_release_request_token = yes
    }

    option = {
        name = "No, cancel"
        ai_chance = { factor = 1 }
        # cancels only its own live request, never a newer one
        if = {
            limit = { ht_annex_request_is_live = yes }
            clr_country_flag = ht_annex_pending
            every_state = { limit = { has_state_flag = ht_live_annex_req } clr_state_flag = ht_live_annex_req }
            clear_variable = target
            clear_variable = taker
            clear_global_event_target = ht_annexed_country
            clear_global_event_target = ht_annexer_country
        }
        # this popup is answered: free its token, live or orphaned
        ht_release_request_token = yes
    }
}

# Host tool confirmation popups: gives the request being opened an identity token, a state saved
# as the LOCAL event target ht_req_token, which the engine copies into the popup fired afterwards;
# the caller flags that state as the live request. A popup may act only while its own token still
# carries the live flag (ht_*_request_is_live), so a popup orphaned by Clear All or a handoff can
# never act on a later request, even one for the same countries.
# A token stays reserved until its own popup is answered (both options release it, live or not),
# so an orphaned popup's token is never handed to a later request. Temp ht_req_token_ok = 1 tells
# the caller a token was issued; if every state is reserved (over a thousand unanswered popups)
# none is, and the caller must not open a popup.
ht_issue_request_token = {
	set_temp_variable = { ht_req_token_ok = 0 }
	random_state = {
		limit = { NOT = { has_state_flag = ht_req_reserved } }
		set_state_flag = ht_req_reserved
		save_event_target_as = ht_req_token
		set_temp_variable = { ht_req_token_ok = 1 }
	}
}

# Popup option, any outcome: frees this popup's own token for reuse.
ht_release_request_token = {
	if = {
		limit = { has_event_target = ht_req_token }
		event_target:ht_req_token = { clr_state_flag = ht_req_reserved }
	}
}
